// Function to request and process detection from Raspberry Pi
bool requestDetection() {
  Serial.println("REQUEST_DETECTION");  // Command to trigger detection on Pi
  
  // Wait for response with timeout
  unsigned long startTime = millis();
  while (millis() - startTime < 2000) { // 2-second timeout
    if (Serial.available() > 0) {
      detectionResult = Serial.readStringUntil('\n');
      detectionResult.trim();
      
      // Parse the result
      if (detectionResult.startsWith("DETECTED:")) {
        Serial.print("Detection result: ");
        Serial.println(detectionResult);
        objectDetected = true;
        return true;
      } 
      else if (detectionResult == "NO_DETECTION") {
        Serial.println("No objects detected");
        objectDetected = false;
        return false;
      }
    }
    delay(10);
  }
  
  Serial.println("Detection request timed out");
  return false;
}

// PID Code
void PID() {
  //Getting the readings of the left and right IR sensor
  int left_result = analogRead(Left_IRSensor);
  int right_result = analogRead(Right_IRSensor);

  // Map the results, white first, black second
  left_result = map(left_result, 59, 656, 0, 100);
  right_result = map(right_result, 65 , 757, 0, 100);

  // Tracking code
  error = left_result - right_result;
  integral += error; // testing
  double derivative = error - lasterror;
  moving = kp * error + ki * integral + kd * derivative;
  turning(150 - moving, 150 + moving);
  lasterror = error;
  delay(30);

}

//Normal Junction Code (for milk and normal junction)
void junction() {
  while (true) {
    int left_result = analogRead(Left_IRSensor);
    int right_result = analogRead(Right_IRSensor);

    if (left_result > 300 and right_result > 300) {
      nomove();
      break;
    } else {
      PID();
    }
  }
}

// Function for backward line following
void PID_Backward() {
  //Getting the readings of the left and right IR sensor
  int left_result = analogRead(Left_IRSensor);
  int right_result = analogRead(Right_IRSensor);

  // Map the results, white first, black second
  left_result = map(left_result, 59, 490, 0, 100);
  right_result = map(right_result, 65, 771, 0, 100);

  // Tracking code (note: error calculation swapped for reverse direction)
  error = right_result - left_result; // Swapped for backward movement
  integral += error;
  double derivative = error - lasterror;
  moving = kp * error + ki * integral + kd * derivative;
  turning(-150 + moving, -150 - moving); // Negative speeds for backward
  lasterror = error;
  delay(30);
}
