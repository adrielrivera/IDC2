// Simplest Arduino detection test

void setup() {
  pinMode(LED_BUILTIN, OUTPUT);  // Built-in LED
  Serial.begin(9600);
  
  // Wait for serial connection
  delay(2000);
  
  // Flash LED to show we're starting
  blinkLED(3);
  
  // Send detection request
  Serial.println("DETECT");
  digitalWrite(LED_BUILTIN, HIGH);  // LED ON = waiting for response
}

void loop() {
  // Check for any response
  if (Serial.available() > 0) {
    String response = Serial.readStringUntil('\n');
    
    // When we receive START, we know detection is beginning
    if (response == "START") {
      digitalWrite(LED_BUILTIN, LOW);  // LED OFF briefly
      delay(100);
      digitalWrite(LED_BUILTIN, HIGH); // LED ON again
    }
    
    // When we receive END, detection is complete
    if (response == "END") {
      digitalWrite(LED_BUILTIN, LOW);  // LED OFF = detection complete
      // Success pattern
      delay(500);
      blinkLED(5); // Blink 5 times quickly to indicate success
    }
  }
}

void blinkLED(int times) {
  for (int i = 0; i < times; i++) {
    digitalWrite(LED_BUILTIN, HIGH);
    delay(100);
    digitalWrite(LED_BUILTIN, LOW);
    delay(100);
  }
}
