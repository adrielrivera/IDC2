// Function to request and handle detection
bool requestDetection() {
  // Clear any existing serial data
  while(Serial.available()) {
    Serial.read();
  }
  
  Serial.println("REQUEST_DETECTION");
  Serial.flush();
  
  // Wait for response with timeout
  unsigned long startTime = millis();
  bool detectionReceived = false;
  
  while (millis() - startTime < 3000) { // 3 second timeout
    if (Serial.available() > 0) {
      String response = Serial.readStringUntil('\n');
      response.trim();
      
      // Check for detection results
      if (response.startsWith("DETECTED_")) {
        String detectedObject = response.substring(9); // Remove "DETECTED_" prefix
        Serial.print("Detected: ");
        Serial.println(detectedObject);
        
        // If hotdog detected, close gate
        if (detectedObject == "HOTDOG") {
          Serial.println("Hotdog detected! Closing gate...");
          gateServo.write(GATE_CLOSED);
          delay(1000);
          gateServo.write(GATE_OPEN);
          detectionReceived = true;
        }
        
        // Echo back to Python
        Serial.print("RECEIVED_");
        Serial.println(detectedObject);
      }
    }
    delay(10);
  }
  
  if (!detectionReceived) {
    Serial.println("No detections received within timeout");
    return false;
  }
  
  return true;
}

// Revised function to parse counts from Python
bool requestAndProcessDetection(bool cycleGateOnAnyTargetFound) {
  // Clear local counts for this new request cycle
  currentSyringeCount = 0;
  currentBandageCount = 0;
  currentGauzeCount = 0;
  detectionCycleProcessed = false; // Reset flag

  // Clear any existing serial data
  while (Serial.available()) {
    Serial.read();
  }

  Serial.println("REQUEST_DETECTION"); // Python script expects this generic command
  Serial.flush();

  unsigned long startTime = millis();
  bool anyTargetItemReportedInThisCycle = false; // True if any COUNT_ for a target item is > 0

  // Loop to receive multiple messages from Python for one detection request
  // This loop will continue until a "processed" or "no target" or "no predictions" message is received, or timeout
  while (millis() - startTime < 5000) { // Increased timeout to 5 seconds to receive all count messages + processed message
    if (Serial.available() > 0) {
      String response = Serial.readStringUntil('\n');
      response.trim();
      Serial.print("Pi -> Arduino: "); Serial.println(response); // Echo raw response for debugging

      if (response.startsWith("COUNT_SYRINGE:")) {
        currentSyringeCount = response.substring(14).toInt(); // "COUNT_SYRINGE:".length()
        Serial.print("Parsed Syringe Count: "); Serial.println(currentSyringeCount);
        if (currentSyringeCount > 0) anyTargetItemReportedInThisCycle = true;
      } else if (response.startsWith("COUNT_BANDAGE:")) {
        currentBandageCount = response.substring(14).toInt(); // "COUNT_BANDAGE:".length()
        Serial.print("Parsed Bandage Count: "); Serial.println(currentBandageCount);
        if (currentBandageCount > 0) anyTargetItemReportedInThisCycle = true;
      } else if (response.startsWith("COUNT_GAUZE:")) {
        currentGauzeCount = response.substring(12).toInt();   // "COUNT_GAUZE:".length()
        Serial.print("Parsed Gauze Count: "); Serial.println(currentGauzeCount);
        if (currentGauzeCount > 0) anyTargetItemReportedInThisCycle = true;
      } else if (response == "DETECTION_PROCESSED") {
        Serial.println("Pi signaled all data sent for this detection cycle.");
        detectionCycleProcessed = true;
        break; // Exit the while loop, we have all info for this detection
      } else if (response == "NO_TARGET_ITEMS_FOUND") {
        Serial.println("Pi reported NO TARGET medical supplies found in this frame.");
        detectionCycleProcessed = true; // Still a valid end to the detection cycle
        anyTargetItemReportedInThisCycle = false;
        break; // Exit loop
      } else if (response == "NO_PREDICTIONS_FROM_MODEL") {
        Serial.println("Pi reported NO PREDICTIONS from model for this frame.");
        detectionCycleProcessed = true; // Still a valid end
        anyTargetItemReportedInThisCycle = false;
        break; // Exit loop
      }
      // Add other general responses from Pi if needed
      
      // Reset startTime for timeout only if we are still expecting more data for *this* detection cycle
      // For example, if we haven't received DETECTION_PROCESSED yet.
      // This prevents timeout if Python sends multiple COUNT messages with small delays.
      startTime = millis(); 
    }
    delay(5); // Small delay in receive loop
  }

  if (!detectionCycleProcessed) {
    Serial.println("TIMEOUT or incomplete data from Pi for detection cycle.");
    return false; // Indicate failure or timeout
  }

  // Now, after processing all messages for this cycle, decide if action is needed
  if (anyTargetItemReportedInThisCycle) {
    Serial.println("One or more target medical supplies were counted.");
    // You can print the final counts:
    Serial.print("Final Counts - Syringes: "); Serial.print(currentSyringeCount);
    Serial.print(", Bandages: "); Serial.print(currentBandageCount);
    Serial.print(", Gauzes: "); Serial.println(currentGauzeCount);

    if (cycleGateOnAnyTargetFound) {
      Serial.println("Cycling gate for collection (as per request)...");
      gateServo.write(GATE_CLOSED);
      delay(1000); // Time for gate to close
      gateServo.write(GATE_OPEN);
      delay(500); // Time for gate to open
    }
    return true; // Indicate a successful detection of at least one target item
  } else {
    Serial.println("No target medical supplies were counted in this cycle.");
    return false; // Indicate no target items found
  }
}

void food() {
  Serial.println("--- Starting Food Collection Routine (Camera Scan, Robot Turn) ---");
  bool foodItemFoundAndCollected = false;
  int detectedAtAngle = -1; // -1 means not found yet

  // Ensure gate is open at the start
  gateServo.write(GATE_OPEN);
  delay(SERVO_MOVE_DELAY); // Allow gate to open

  int scanAngles[] = {CAMERA_POSITION_CENTER, CAMERA_POSITION_LEFT_SCAN, CAMERA_POSITION_RIGHT_SCAN};
  // Or you might prefer an order like: LEFT, CENTER, RIGHT

  for (int i = 0; i < (sizeof(scanAngles) / sizeof(scanAngles[0])); i++) {
    int currentScanAngle = scanAngles[i];
    Serial.print("Moving camera to scan angle: ");
    Serial.println(currentScanAngle);
    cameraServo.write(currentScanAngle);
    delay(SERVO_MOVE_DELAY);

    Serial.println("Requesting detection...");
    if (requestDetection()) { // Assumes requestDetection returns true for HOTDOG
      detectedAtAngle = currentScanAngle;
      Serial.print("HOTDOG detected by camera at angle: ");
      Serial.println(detectedAtAngle);
      break; // Exit scan loop
    } else {
      Serial.println("No HOTDOG detected at this camera angle.");
    }
  }
  
  // --- Robot Alignment and Collection Phase ---
  if (detectedAtAngle != -1) {
    Serial.println("Proceeding to robot alignment and collection.");

    // 1. Align Robot Body
    if (detectedAtAngle == CAMERA_POSITION_LEFT_SCAN) {
      Serial.println("Aligning robot: Turning LEFT.");
      turning(-ROBOT_TURN_SPEED, ROBOT_TURN_SPEED); // Pivot left
      delay(ROBOT_TURN_DURATION_30_DEG);
      nomove();
      delay(ROBOT_SETTLE_DELAY);
    } else if (detectedAtAngle == CAMERA_POSITION_RIGHT_SCAN) {
      Serial.println("Aligning robot: Turning RIGHT.");
      turning(ROBOT_TURN_SPEED, -ROBOT_TURN_SPEED); // Pivot right
      delay(ROBOT_TURN_DURATION_30_DEG);
      nomove();
      delay(ROBOT_SETTLE_DELAY);
    } else if (detectedAtAngle == CAMERA_POSITION_CENTER) {
      Serial.println("Robot already aligned (food detected at center camera).");
    }

    // 2. Collection Maneuver (gate was already cycled to OPEN by requestDetection)
    Serial.println("Performing collection: Nudge forward & Secure gate.");
    moveforward(COLLECTION_NUDGE_SPEED);
    delay(COLLECTION_NUDGE_DURATION);
    nomove();
    gateServo.write(GATE_CLOSED); // Secure the item
    delay(SERVO_MOVE_DELAY);    // Let gate close
    

    moveback(150);
    delay(8000);
    foodItemFoundAndCollected = true;

  } else {
    Serial.println("HOTDOG NOT detected in any camera scan position.");
  }

  // --- Post-Routine Actions ---
  // Optional: Return camera to center
  cameraServo.write(CAMERA_POSITION_CENTER);
  delay(SERVO_MOVE_DELAY);

  if (foodItemFoundAndCollected) {
    Serial.println("SUCCESS: Food item collection process complete!");
    // Prepare for next task, e.g., move away from food station
  } else {
    Serial.println("FAILURE: Food item not found or collected.");
    // Proceed to alternative task or signal failure
  }
  Serial.println("--- Food Collection Routine Finished ---");
}

// PID Code
void PID() {
  //Getting the readings of the left and right IR sensor
  int left_result = analogRead(Left_IRSensor);
  int right_result = analogRead(Right_IRSensor);

  // Map the results, white first, black second
  left_result = map(left_result, 59, 656, 0, 100);
  right_result = map(right_result, 65 , 757, 0, 100);

  // Tracking code
  error = left_result - right_result;
  integral += error; // testing
  double derivative = error - lasterror;
  moving = kp * error + ki * integral + kd * derivative;
  turning(150 - moving, 150 + moving);
  lasterror = error;
  delay(30);

}

      nomove();
      break;
    } else {
      PID();
    }
  }
}
